CREATE DATABASE IF NOT EXISTS `factoriocommunity`;
USE `factoriocommunity`;

DROP TABLE IF EXISTS `Files`;
DROP TABLE IF EXISTS `Comments`;
DROP TABLE IF EXISTS `Posts`;
DROP TABLE IF EXISTS `Users`;

-- Users
CREATE TABLE Users (
  username VARCHAR(63) NOT NULL,
  password VARCHAR(255) NOT NULL,
  name VARCHAR(63) NOT NULL,
  creation_date DATETIME NOT NULL,
  profile_description MEDIUMTEXT,
  profile_picture VARCHAR(255),
  PRIMARY KEY (username)
);

-- Posts
CREATE TABLE Posts (
  post_id INT NOT NULL AUTO_INCREMENT,
  creation_timestamp DATETIME NOT NULL,
  title VARCHAR(255) NOT NULL,
  body MEDIUMTEXT NOT NULL,
  Users_username VARCHAR(63) NOT NULL,
  PRIMARY KEY (post_id),
  FOREIGN KEY (Users_username) REFERENCES Users (username)
);

-- Comments
CREATE TABLE Comments (
  comment_id INT NOT NULL AUTO_INCREMENT,
  body VARCHAR(4095) NOT NULL,
  creation_timestamp DATETIME NOT NULL,
  Posts_post_id INT NOT NULL,
  Users_username VARCHAR(63) NOT NULL,
  PRIMARY KEY (comment_id),
  FOREIGN KEY (Posts_post_id) REFERENCES Posts (post_id),
  FOREIGN KEY (Users_username) REFERENCES Users (username)
);
