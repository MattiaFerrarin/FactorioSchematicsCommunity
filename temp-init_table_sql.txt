-- Users
CREATE TABLE Users (
  username VARCHAR(63) NOT NULL,
  password VARCHAR(63) NOT NULL,
  name VARCHAR(63) NOT NULL,
  creation_date DATETIME NOT NULL,
  profile_description MEDIUMTEXT NOT NULL,
  profile_picture VARCHAR(255) NOT NULL,
  PRIMARY KEY (username)
)

-- Posts
CREATE TABLE Posts (
  post_id VARCHAR(255) NOT NULL AUTO_INCREMENT,
  creation_timestamp DATETIME NOT NULL,
  title VARCHAR(255) NOT NULL,
  body MEDIUMTEXT NOT NULL,
  Users_username INT NOT NULL, -- here
  PRIMARY KEY (post_id),
  FOREIGN KEY (Users_username) REFERENCES Users (username)
)

-- Comments
CREATE TABLE Comments (
  comment_id VARCHAR(255) NOT NULL,
  body VARCHAR(255) NOT NULL,
  timestamp VARCHAR(255) NOT NULL,
  Posts_post_id INT NOT NULL,
  Users_username INT NOT NULL,
  PRIMARY KEY (comment_id),
  FOREIGN KEY (Posts_post_id) REFERENCES Posts (post_id),
  FOREIGN KEY (Users_username) REFERENCES Users (username)
)
